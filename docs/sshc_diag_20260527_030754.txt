    function fmtBytes(n) {
      if (!n) return '--';
      if (n >= 1073741824) return (n / 1073741824).toFixed(1) + ' GB';
      if (n >= 1048576) return (n / 1048576).toFixed(1) + ' MB';
      if (n >= 1024) return (n / 1024).toFixed(0) + ' KB';
      return n + ' B';
    }
    function fmtUptime(s) {
      s = Number(s || 0);
      const h = Math.floor(s / 3600);
      const m = Math.floor((s % 3600) / 60);
      const sec = s % 60;
      if (h) return h + 'h ' + m + 'm';
      if (m) return m + 'm ' + sec + 's';
      return sec + 's';
    }
    function ipMeta(d) {

      const net = [d.isp, d.asn].filter(Boolean).join(' - ') || 'unknown network';
      return loc + ' / ' + net + (d.cached ? ' / cached' : '');
    }

      } else if (busy) {
        $('statusDot').className = 'status-dot connecting';
        $('statusText').textContent = 'connecting';
      } else {

        $('statusText').textContent = 'disconnected';

    async function saveProfile(select, restart) {
      try {
        const mode = $('pfMode').value;
        const usesProxy = mode.includes('http_proxy');
        const usesTLS = mode.includes('tls');
        const req = {
          id: profileEditId,
          name: $('pfName').value.trim() || 'SSHCustom Profile',
          select,
          restart,
          ssh: {
            host: $('pfHost').value.trim(),
            port: parseInt($('pfPort').value, 10) || 22,
            username: $('pfUser').value.trim(),
            password: $('pfPass').value,
            auth_type: 'password'
            // fallback_ips removed in v2.0.0: hostnames resolve through the
            // daemon's smart resolver. If a profile previously had fallback_ips,
            // the daemon keeps them intact in profiles.json on save.
          },
          transport: {
            mode,
            http_proxy: usesProxy ? {
              host: $('pfProxyHost').value.trim() || $('pfHost').value.trim(),
              port: parseInt($('pfProxyPort').value, 10) || 80,
              connect_method: 'socket'
            } : undefined,
            tls: usesTLS ? {
              enabled: true,
              server_name: $('pfSni').value.trim() || $('pfHost').value.trim(),
              insecure_skip_verify: true,
              alpn: ['http/1.1']
            } : undefined,
            payload: {
              enabled: $('payloadSwitch').classList.contains('on'),
              template: $('pfPayload').value,
              send_timing: usesProxy ? 'after_proxy_socket_before_ssh' : 'before_ssh',
              read_response: true,
              allow_http_status: [101, 200, 204, 302]
            }
          }
        };
        await postV1('/profile/save', req, 12000);
        closeProfileEditor();
        if (restart) {
          toast('Profile saved. Restarting tunnel...');
          showRestarting();
          setTimeout(loadStatus, 3000);
        } else {
          toast('Profile saved.');
        }
        await loadProfiles();
      } catch (e) {
        toast('Profile save failed: ' + e.message, 'error');
      }
    }

    async function vcLoadLocations() {
      try {
        const data = await v1('/vpnchain/locations');
        vcLocations = data || [];
        const sel = $('vcLocation');
        if (vcLocations.length === 0) {
          sel.innerHTML = '<option value="">No configs found</option>';
        } else {
          sel.innerHTML = vcLocations.map(l => `<option value="${esc(l)}">${esc(l)}</option>`).join('');
          // Restore last selected location from localStorage
          const saved = localStorage.getItem('vcSelectedLocation');
          if (saved && vcLocations.includes(saved)) {
            sel.value = saved;
          } else if (vcState.running && vcState.location && vcLocations.includes(vcState.location)) {
            sel.value = vcState.location;
          }
        }
      } catch (e) {
        $('vcLocation').innerHTML = '<option value="">Error loading</option>';
      }

    // --- REFRESH ENGINE ---
    // Primary: reliable polling loop (like the proven old version).
    // Enhancement: SSE for instant status updates when connected.
    // The polling loop ALWAYS runs — SSE just makes status updates faster.
    let tick = 0;
    let pollTimer = null;
    let sseSource = null;

    function refresh() {
      tick++;
      loadStatus();
      // Refresh page-specific content on slower cadences
      if (currentPage === 'profiles' && tick % 4 === 0) loadProfiles();
      if (currentPage === 'runtime') {
        if (tick % 3 === 0) loadDiagnostics();
        if (tick % 4 === 0) loadLog();
      }
      if (currentPage === 'vpnchain' && tick % 3 === 0) vcLoadStatus();
      if (tick % 20 === 0) loadPublicIP(false);
    }

    function startPolling() {
      if (pollTimer) return;
      pollTimer = setInterval(refresh, 3000);
    }
    function stopPolling() {
      if (pollTimer) { clearInterval(pollTimer); pollTimer = null; }
    }

    function startEvents() {
      if (sseSource) { try { sseSource.close(); } catch(_){} sseSource = null; }
      try {
        sseSource = new EventSource(API + '/api/v1/events');
      } catch (e) { return; }
      sseSource.addEventListener('status', (ev) => {
        try {
          const data = JSON.parse(ev.data);
          statusData = data.runtime;
          configData = data.config;
          pathsData = data.paths || {};
          renderStatus(data);
          renderPaths();
          renderAbout();
        } catch (_) {}
      });
      sseSource.onerror = () => {
        // SSE died (daemon restarted, network blip, etc.) — close and retry in 5s.
        // Polling keeps running regardless so the UI never freezes.
        try { sseSource.close(); } catch(_) {}
        sseSource = null;
        setTimeout(startEvents, 5000);
      };
    }

    // Always start polling + SSE together. Polling guarantees freshness;
    // SSE makes it instant when it's working.
    startPolling();
    startEvents();
  </script>
